Declaration-First Syntax.

Orthogonal grammar rules and explicit method receivers ensure zero ambiguity between declarations and expressions. 

import std.io: print

Point :: struct {
    x, y i32

    dist_sq :: fn(self &Self) i32 {
        self.x * self.x + self.y * self.y
    }
}

main :: fn() {
    p: Point = .{ x: 3, y: 4 }
    print(p.dist_sq())
}
